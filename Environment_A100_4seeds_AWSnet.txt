#!/bin/bash
# ==============================================================================
# Script cài đặt môi trường và Huấn luyện SONG SONG 4 SEEDS tối ưu cho NVIDIA A100
# Mô hình: AWSnet (Attention U-Net with Auto-Weighted Deep Supervision, 8.73M params)
# GPU: NVIDIA A100 (40GB / 80GB VRAM) trên Google Colab / Cloud
# Repository: https://github.com/mintduckk24/AWSnet.git
# 4 Seeds chuẩn hóa theo yêu cầu: [6868, 1024, 1098, 2026]
# Quản lý: 4 Terminal độc lập thông qua tmux (Session: train4_awsnet)
# Benchmark hỗ trợ:
#   1. MyoPS++: Profile myopspp_unet2d_roi128_1mm (64 train / 16 test / no-val / HD95 tính bằng mm)
#   2. MyoPS380: Protocol myops380_train304_test76_last_v1 (304 train / 76 test / no-val / HD95 tính bằng voxel)
# ==============================================================================

set -e

echo "=== [1/7] Kiểm tra và Kích hoạt GPU NVIDIA A100 ==="
cd /content

# Tự động liên kết thư viện driver NVIDIA nếu có trên Colab (/usr/lib64-nvidia)
if [ -d "/usr/lib64-nvidia" ]; then
    if ! grep -q "/usr/lib64-nvidia" /etc/ld.so.conf.d/nvidia.conf 2>/dev/null; then
        echo "/usr/lib64-nvidia" >> /etc/ld.so.conf.d/nvidia.conf
        ldconfig 2>/dev/null || true
    fi
    export LD_LIBRARY_PATH=/usr/lib64-nvidia:${LD_LIBRARY_PATH}
fi
if ! grep -q "/usr/lib64-nvidia" ~/.bashrc 2>/dev/null; then
    echo 'export LD_LIBRARY_PATH=/usr/lib64-nvidia:$LD_LIBRARY_PATH' >> ~/.bashrc 2>/dev/null || true
fi

if ! command -v nvidia-smi &> /dev/null; then
    echo "⚠️ CẢNH BÁO: Chưa tìm thấy NVIDIA GPU. Hãy kiểm tra lại Runtime > Change runtime type > A100 GPU trên Colab!"
else
    echo "Thông tin GPU hiện tại:"
    nvidia-smi --query-gpu=name,memory.total,driver_version --format=csv,noheader || true
fi

# Tối ưu hóa cấp phát bộ nhớ CUDA cho A100 (chống phân mảnh khi chạy song song 4 tiến trình)
export PYTORCH_CUDA_ALLOC_CONF="expandable_segments:True"

echo "=== [2/7] Mount Google Drive (Lưu Checkpoints & Logs an toàn) ==="
python3 -c "
import os
if not os.path.exists('/content/drive/MyDrive'):
    try:
        from google.colab import drive
        drive.mount('/content/drive')
        print('✅ Mount Google Drive thành công!')
    except Exception as e:
        print('⚠️ Không thể mount Drive tự động (bỏ qua nếu không dùng):', e)
" || true

echo "=== [3/7] Thiết lập Repository AWSnet ==="
REPO_URL="https://github.com/mintduckk24/AWSnet.git"

if [ -f "./train.py" ] && [ -d "./models" ]; then
    echo "✅ Đang đứng trực tiếp trong thư mục AWSnet."
elif [ -d "/content/AWSnet" ]; then
    echo "✅ Thư mục /content/AWSnet đã tồn tại."
elif [ -f "/content/AWSnet.zip" ]; then
    echo "✅ Phát hiện file AWSnet.zip tại /content/. Đang giải nén..."
    unzip -q -o "/content/AWSnet.zip" -d /content/
elif [ -f "/content/drive/MyDrive/AWSnet.zip" ]; then
    echo "✅ Phát hiện file AWSnet.zip trên Google Drive. Đang giải nén..."
    unzip -q -o "/content/drive/MyDrive/AWSnet.zip" -d /content/
elif [ -d "/content/drive/MyDrive/AWSnet" ]; then
    echo "✅ Phát hiện thư mục AWSnet trên Google Drive. Đang sao chép vào /content/AWSnet..."
    cp -r "/content/drive/MyDrive/AWSnet" /content/AWSnet
else
    echo "Đang clone repository từ $REPO_URL..."
    git clone "$REPO_URL" /content/AWSnet || {
        echo "❌ Chưa thể clone từ $REPO_URL."
        echo "👉 Bạn có thể kéo thả file AWSnet.zip vào khung Explorer của VS Code (/content/) rồi chạy lại."
        exit 1
    }
fi

if [ -d "/content/AWSnet" ]; then
    cd /content/AWSnet
fi
AWSNET_DIR="/content/AWSnet"

echo "=== [4/7] Cài đặt Dependencies cho AWSnet & tmux ==="
python3 -m pip install --upgrade pip -q
if [ -f "requirements.txt" ]; then
    python3 -m pip install -r requirements.txt gdown -q
else
    python3 -m pip install torch torchvision numpy scipy h5py nibabel pandas tqdm gdown -q
fi
command -v tmux &> /dev/null || (apt-get update -qq && apt-get install -y tmux)

# Cấu hình tmux hỗ trợ con lăn chuột và hiển thị 4 tab
cat << 'EOF' > /root/.tmux.conf
set -g mouse on
set -g base-index 1
setw -g pane-base-index 1
set -g status-bg colour235
set -g status-fg colour136
set -g status-left '#[fg=green][A100-AWSnet-4Seeds] '
set -g status-right '#[fg=yellow]%H:%M:%S'
EOF

echo "=== [5/7] Chuẩn bị Dataset (Hỗ trợ cả MyoPS++ và MyoPS380) ==="
# Cấu hình Dataset mặc định: myopspp_unet2d_roi128_1mm (hoặc myops380)
DATASET_NAME="${DATASET_NAME:-myopspp_unet2d_roi128_1mm}"

if [ "$DATASET_NAME" = "myopspp_unet2d_roi128_1mm" ]; then
    echo "🎯 Lựa chọn Dataset: MyoPS++ (ROI 128x128, Spacing 1.0mm, Đơn vị HD95: mm)"
    MYOPSPP_DIR="/content/AWSnet/Dataset/MyoPSpp_UNet2D_ROI128_1mm"
    MYOPSPP_ZIP="/content/AWSnet/Dataset/MyoPSpp_UNet2D_ROI128_1mm.zip"
    MYOPSPP_ID="1FmHR9_e0AW3snN_qscUsVGn7Vk0y3fLk"

    # Kiểm tra dataset cục bộ hoặc trên Drive
    MYOPSPP_SRC=$(find /content/drive/MyDrive /content/AWSnet/Dataset /content/Dataset /content -name "dataset_manifest.json" -exec dirname {} \; 2>/dev/null | head -n 1 || true)
    if [ -n "$MYOPSPP_SRC" ] && [ -d "$MYOPSPP_SRC/cases" ]; then
        LOCAL_DATASET="$MYOPSPP_SRC"
    elif [ -d "$MYOPSPP_DIR/cases" ]; then
        LOCAL_DATASET="$MYOPSPP_DIR"
    elif [ -d "$MYOPSPP_DIR/MyoPSpp_UNet2D_ROI128_1mm/cases" ]; then
        LOCAL_DATASET="$MYOPSPP_DIR/MyoPSpp_UNet2D_ROI128_1mm"
    else
        mkdir -p "$MYOPSPP_DIR"
        if [ ! -s "$MYOPSPP_ZIP" ]; then
            echo "⬇️ Tải MyoPSpp_UNet2D_ROI128_1mm qua gdown..."
            gdown --id "$MYOPSPP_ID" -O "$MYOPSPP_ZIP" || true
        fi
        if [ -s "$MYOPSPP_ZIP" ]; then
            unzip -q -o "$MYOPSPP_ZIP" -d "$MYOPSPP_DIR/"
        fi
        MYOPSPP_SRC=$(find "$MYOPSPP_DIR" -name "dataset_manifest.json" -exec dirname {} \; 2>/dev/null | head -n 1 || true)
        if [ -n "$MYOPSPP_SRC" ] && [ -d "$MYOPSPP_SRC/cases" ]; then
            LOCAL_DATASET="$MYOPSPP_SRC"
        elif [ -d "$MYOPSPP_DIR/cases" ]; then
            LOCAL_DATASET="$MYOPSPP_DIR"
        elif [ -d "$MYOPSPP_DIR/MyoPSpp_UNet2D_ROI128_1mm/cases" ]; then
            LOCAL_DATASET="$MYOPSPP_DIR/MyoPSpp_UNet2D_ROI128_1mm"
        else
            LOCAL_DATASET="$MYOPSPP_DIR"
        fi
    fi

    SPLIT_DIR="$AWSNET_DIR/dataset/splits/myopspp_unet2d_roi128_1mm"
    LABEL_ORDER="canonical"
    DISTANCE_UNIT="mm"
    echo "✅ Đã xác định Dataset MyoPS++: $LOCAL_DATASET (Split: $SPLIT_DIR, Nhãn: $LABEL_ORDER, HD95: $DISTANCE_UNIT)"

else
    echo "🎯 Lựa chọn Dataset: MyoPS380 (Chuẩn Legacy, Đơn vị HD95: voxel)"
    DATASET_FOLDER_ID="13T6892xy-pUwC1BSjGl4sRxoypMu-SIS"
    DOWNLOAD_DIR="/content/dataset_download"
    TARGET_DATASET_DIR="/content/AWSnet/MyoPS380_dataset"

    mkdir -p "$DOWNLOAD_DIR"
    mkdir -p "$TARGET_DATASET_DIR"

    DRIVE_DATASET_DIR="/content/drive/MyDrive/MyoPS380_dataset"
    if [ -d "$DRIVE_DATASET_DIR/Processed_data/bSSFP" ]; then
        LOCAL_DATASET="$DRIVE_DATASET_DIR/Processed_data"
    elif [ -d "/content/drive/MyDrive/Processed_data/bSSFP" ]; then
        LOCAL_DATASET="/content/drive/MyDrive/Processed_data"
    else
        if [ -z "$(ls -A "$DOWNLOAD_DIR" 2>/dev/null)" ] && [ -z "$(ls -A "$TARGET_DATASET_DIR" 2>/dev/null)" ]; then
            echo "Đang tải dataset MyoPS380 từ Google Drive..."
            gdown --folder --id "$DATASET_FOLDER_ID" -O "$DOWNLOAD_DIR" --remaining-ok || true
        fi
        for zipfile in $(find "$DOWNLOAD_DIR" -maxdepth 2 -name "*.zip" 2>/dev/null); do
            unzip -q -o "$zipfile" -d "$TARGET_DATASET_DIR/"
        done
        FOUND_TRAIN_NPZ=$(find "$TARGET_DATASET_DIR" "$DOWNLOAD_DIR" /content -type d -name "train_npz" 2>/dev/null | grep -i "bSSFP" | head -n 1 || true)
        if [ -n "$FOUND_TRAIN_NPZ" ]; then
            LOCAL_DATASET=$(dirname $(dirname "$FOUND_TRAIN_NPZ"))
        else
            LOCAL_DATASET="/content/AWSnet/MyoPS380_dataset/Processed_data"
        fi
    fi

    SPLIT_DIR="$AWSNET_DIR/dataset/splits/myops380_no_val"
    LABEL_ORDER="legacy"
    DISTANCE_UNIT="voxel"
    echo "✅ Đã xác định Dataset MyoPS380: $LOCAL_DATASET (Split: $SPLIT_DIR, Nhãn: $LABEL_ORDER, HD95: $DISTANCE_UNIT)"
fi

echo "=== [6/7] Thiết lập Cấu hình Huấn luyện Tối ưu cho 4 Seeds Song Song ==="
if [ -d "/content/drive/MyDrive" ]; then
    RUN_ROOT="/content/drive/MyDrive/AWSnet_Runs"
else
    RUN_ROOT="/content/AWSnet_Runs"
fi
LOGS_DIR="$RUN_ROOT/parallel_logs_awsnet_4seeds"
mkdir -p "$LOGS_DIR"

# Danh sách 4 Seeds chính xác theo yêu cầu: [6868, 1024, 1098, 2026]
SEEDS=(6868 1024 1098 2026)

echo "Mô hình: AWSnet (Attention U-Net with Auto-Weighted Deep Supervision, 8.73M params)"
echo "4 Seeds khởi chạy đồng thời: ${SEEDS[*]}"
echo "Thư mục lưu Checkpoints: $RUN_ROOT"
echo "Thư mục lưu Logs: $LOGS_DIR"
echo "Đơn vị khoảng cách HD95 khi đánh giá: $DISTANCE_UNIT"

echo "=== [7/7] Khởi động 4 Terminal TMUX chạy 4 Seeds Song Song ==="

tmux kill-session -t train4_awsnet 2>/dev/null || true

# Tạo session tmux mới với terminal đầu tiên (Seed 6868)
FIRST_SEED=${SEEDS[0]}
FIRST_LOG="$LOGS_DIR/train_awsnet_seed_${FIRST_SEED}.log"

echo "🚀 [1/4] Khởi động Terminal 1 (Seed $FIRST_SEED)..."
tmux new-session -d -s train4_awsnet -n "awsnet_${FIRST_SEED}" \
    "cd $AWSNET_DIR && export PYTORCH_CUDA_ALLOC_CONF=expandable_segments:True && \
     python3 train.py \
        --data_dir '$LOCAL_DATASET' \
        --split_dir '$SPLIT_DIR' \
        --save_dir '$RUN_ROOT' \
        --dataset '$DATASET_NAME' \
        --label_order '$LABEL_ORDER' \
        --seed $FIRST_SEED \
        --batch_size 16 \
        --accum_steps 1 \
        --max_epochs 300 \
        --base_lr 0.0003333 \
        --weight_decay 0.0001 \
        --num_workers 4 \
        --amp auto \
        --resume 2>&1 | tee '$FIRST_LOG'"

# Khởi tạo tiếp 3 terminal (window) cho 3 seeds còn lại
for i in {1..3}; do
    SEED=${SEEDS[$i]}
    INDEX=$((i + 1))
    LOG_FILE="$LOGS_DIR/train_awsnet_seed_${SEED}.log"
    echo "🚀 [$INDEX/4] Khởi động Terminal $INDEX (Seed $SEED)..."

    tmux new-window -t train4_awsnet -n "awsnet_${SEED}" \
        "cd $AWSNET_DIR && export PYTORCH_CUDA_ALLOC_CONF=expandable_segments:True && \
         python3 train.py \
            --data_dir '$LOCAL_DATASET' \
            --split_dir '$SPLIT_DIR' \
            --save_dir '$RUN_ROOT' \
            --dataset '$DATASET_NAME' \
            --label_order '$LABEL_ORDER' \
            --seed $SEED \
            --batch_size 16 \
            --accum_steps 1 \
            --max_epochs 300 \
            --base_lr 0.0003333 \
            --weight_decay 0.0001 \
            --num_workers 4 \
            --amp auto \
            --resume 2>&1 | tee '$LOG_FILE'"
done

echo ""
echo "=============================================================================="
echo "🎉 ĐÃ KHỞI CHẠY THÀNH CÔNG 4 SEEDS AWSNET SONG SONG TRÊN 4 TERMINAL TMUX!"
echo "   Seeds: 6868, 1024, 1098, 2026"
echo "   Dataset: $DATASET_NAME | Đơn vị HD95: $DISTANCE_UNIT"
echo "=============================================================================="
echo ""
echo "💡 CÁCH XEM VÀ CHUYỂN ĐỔI GIỮA 4 TERMINAL TRÊN COLAB CLI:"
echo "  1. Kết nối giao diện tmux 4 terminal: tmux attach -t train4_awsnet"
echo "  2. Phím tắt chuyển cửa sổ: Ctrl + B sau đó bấm 1, 2, 3, hoặc 4"
echo "  3. Tách phiên (detach): Bấm Ctrl + B sau đó bấm D"
echo "  4. Theo dõi log thời gian thực: tail -f $LOGS_DIR/train_awsnet_seed_*.log"
echo ""
echo "=============================================================================="
echo "📊 LỆNH ĐÁNH GIÁ VÀ TỔNG HỢP SAU KHI TRAIN XONG (EVALUATION & AGGREGATION):"
echo "=============================================================================="
echo "# Chạy đánh giá cho từng seed (HD95 tự động tính chuẩn $DISTANCE_UNIT):"
for s in "${SEEDS[@]}"; do
    echo "python3 evaluate.py --checkpoint '$RUN_ROOT/awsnet_seed${s}.pt' --data_dir '$LOCAL_DATASET' --split_dir '$SPLIT_DIR' --dataset '$DATASET_NAME' --output_dir '$RUN_ROOT/eval_awsnet_seed${s}' --batch_size 8 --amp auto"
done
echo ""
echo "# Tổng hợp kết quả trung bình 4 seeds:"
echo "python3 -c \"
import json, glob, numpy as np
from pathlib import Path
root = Path('$RUN_ROOT')
dice_myo, dice_edema, dice_scar = [], [], []
hd95_scar, hd95_edema = [], []
for s in [6868, 1024, 1098, 2026]:
    p = root / f'eval_awsnet_seed{s}/metrics.json'
    if p.exists():
        d = json.loads(p.read_text(encoding='utf-8'))
        sm = d.get('summary', {})
        if 'normal_myocardium' in sm: dice_myo.append(sm['normal_myocardium']['mean_dice'])
        if 'edema' in sm: dice_edema.append(sm['edema']['mean_dice'])
        if 'scar' in sm:
            dice_scar.append(sm['scar']['mean_dice'])
            hd95_key = [k for k in sm['scar'] if 'hd95' in k and sm['scar'][k] is not None]
            if hd95_key: hd95_scar.append(sm['scar'][hd95_key[0]])
print('=== KẾT QUẢ TRUNG BÌNH 4 SEEDS AWSNET ===')
if dice_scar: print(f'Scar Dice:  {np.mean(dice_scar):.4f} +/- {np.std(dice_scar):.4f}')
if dice_edema: print(f'Edema Dice: {np.mean(dice_edema):.4f} +/- {np.std(dice_edema):.4f}')
if dice_myo: print(f'Myo Dice:   {np.mean(dice_myo):.4f} +/- {np.std(dice_myo):.4f}')
if hd95_scar: print(f'Scar HD95 ($DISTANCE_UNIT): {np.mean(hd95_scar):.4f} +/- {np.std(hd95_scar):.4f}')
\""
echo "=============================================================================="
